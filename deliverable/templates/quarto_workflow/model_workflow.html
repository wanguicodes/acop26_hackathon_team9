train_arm <- train_full |>
  filter(treatment_arm == arm) |>
  select(all_of(c(predictor_cols, outcome_col))) |>
  rename(outcome = all_of(outcome_col))

test_arm <- test_full |>
  filter(treatment_arm == arm) |>
  select(all_of(c(predictor_cols, outcome_col))) |>
  rename(outcome = all_of(outcome_col))

# Patient IDs for the predictions export (kept out of the modeling data itself)

set.seed(2026)
folds <- vfold_cv(train_arm, v = 5)
metrics <- metric_set(rmse, mae, rsq)

Covariate baseline: Elastic Net

rec_enet <- recipe(outcome ~ ., data = train_arm) |>
  step_novel(all_nominal_predictors()) |>
  step_unknown(all_nominal_predictors()) |>
  step_dummy(all_nominal_predictors()) |>
  step_zv(all_predictors()) |>
  step_normalize(all_numeric_predictors())

spec_enet <- linear_reg(penalty = tune(), mixture = tune()) |>
  set_engine("glmnet")

wf_enet <- workflow() |> add_recipe(rec_enet) |> add_model(spec_enet)

tune_enet <- tune_grid(wf_enet, resamples = folds, metrics = metrics, grid = 5)
best_enet <- select_best(tune_enet, metric = "rmse")

final_enet <- finalize_workflow(wf_enet, best_enet) |> fit(train_arm)

ML stress test: Random Forest

rec_rf <- recipe(outcome ~ ., data = train_arm) |>
  step_novel(all_nominal_predictors()) |>
  step_unknown(all_nominal_predictors()) |>
  step_zv(all_predictors())

spec_rf <- rand_forest(trees = tune(), min_n = tune()) |>
  set_engine("ranger", importance = "permutation", seed = 2026) |>
  set_mode("regression")

wf_rf <- workflow() |> add_recipe(rec_rf) |> add_model(spec_rf)

tune_rf <- tune_grid(wf_rf, resamples = folds, metrics = metrics, grid = 5)
best_rf <- select_best(tune_rf, metric = "rmse")

final_rf <- finalize_workflow(wf_rf, best_rf) |> fit(train_arm)

Test-set evaluation (once, locked)

pred_enet <- predict(final_enet, test_arm)$.pred
pred_rf   <- predict(final_rf, test_arm)$.pred

score_one <- function(obs, pred) {
  tibble(obs = obs, pred = pred) |>
    summarise(
      rmse = sqrt(mean((obs - pred)^2)),
      mae  = mean(abs(obs - pred)),
      r2   = suppressWarnings(cor(obs, pred)^2)
    )
}

scoreboard <- bind_rows(
  score_one(test_arm$outcome, pred_enet) |>
    mutate(arm = arm, method = "Elastic Net", method_type = "covariate", tuned = TRUE),
  score_one(test_arm$outcome, pred_rf) |>
    mutate(arm = arm, method = "Random Forest", method_type = "ml", tuned = TRUE)
) |>
  select(arm, method, method_type, tuned, rmse, mae, r2)
